cmake_minimum_required(VERSION 3.16)
project(ble_scanner_test LANGUAGES C)
enable_testing()
add_executable(ble_scanner_test scanner_test.c ../../src/ble_scanner.c)
target_include_directories(ble_scanner_test PRIVATE stubs ../../src)
target_compile_features(ble_scanner_test PRIVATE c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ble_scanner_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME ble_scanner_cancellation COMMAND ble_scanner_test)
add_test(NAME ble_scanner_first_init COMMAND ble_scanner_test first-init)
set_tests_properties(ble_scanner_cancellation ble_scanner_first_init PROPERTIES TIMEOUT 5)

foreach(variant legacy secure)
  add_executable(ble_uuid_${variant}_test uuid_test.c ../../src/ble_scanner.c)
  target_include_directories(ble_uuid_${variant}_test PRIVATE stubs ../../src)
  target_compile_features(ble_uuid_${variant}_test PRIVATE c_std_11)
  if(variant STREQUAL secure)
    target_compile_definitions(ble_uuid_${variant}_test PRIVATE CONFIG_NORDIC_SECURE_DFU=1)
  endif()
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ble_uuid_${variant}_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME ble_uuid_${variant} COMMAND ble_uuid_${variant}_test)
  set_tests_properties(ble_uuid_${variant} PROPERTIES TIMEOUT 5)
endforeach()
